        <!-- Lời nhắn gõ chữ tự động từ Tuấn Anh -->
        <p class="intro-text" id="typedText"></p>

        <div class="btn-group">
            <button id="yesBtn" onclick="celebrate()">Đồng ý luôn! ❤️</button>
            <button id="noBtn" onmouseover="moveButton()">Không nha 😢</button>

    <script>
        // 1. Lời nhắn đã cá nhân hóa tên Tuấn Anh
        const textToType = "Mình là Tuấn Anh đây. Chỉ mất 1 phút để cậu lướt qua hết trang web này, nhưng mình đã mất rất nhiều thời gian để đứng đây chờ cậu... Cậu có đồng ý bước vào thế giới của mình không?";
        let index = 0;
        
        function typeWriter() {
            if (index < textToType.length) {
                document.getElementById("typedText").innerHTML += textToType.charAt(index);
                index++;
                setTimeout(typeWriter, 50);
            }
        }
        window.onload = typeWriter;

        // 2. Hiệu ứng tạo trái tim rơi
        function createHeart() {
            const heart = document.createElement('div');
            heart.classList.add('heart');
            heart.innerHTML = '❤️';
            heart.style.left = Math.random() * 100 + 'vw';
            heart.style.animationDuration = Math.random() * 3 + 2 + 's';
            heart.style.fontSize = Math.random() * 15 + 15 + 'px';
            document.body.appendChild(heart);

            setTimeout(() => {
                heart.remove();
            }, 5000);
        }
        setInterval(createHeart, 300);

        // 3. Nút "Không" tự động trốn khi di chuột vào
        function moveButton() {
            const noBtn = document.getElementById('noBtn');
            const x = Math.random() * (window.innerWidth - noBtn.offsetWidth - 20);
            const y = Math.random() * (window.innerHeight - noBtn.offsetHeight - 20);
            
            noBtn.style.position = 'fixed';
            noBtn.style.left = x + 'px';
            noBtn.style.top = y + 'px';
        }

        // 4. Kịch bản khi crush nhấn "Đồng ý"
        function celebrate() {
            document.getElementById('title').innerHTML = "Yassss! Mình biết ngay mà! 🎉";
            document.getElementById('typedText').innerHTML = "Cảm ơn cậu vì đã đồng ý nhé! Hãy chụp màn hình giao diện này lại và gửi ngay cho Tuấn Anh nhaaa. Chờ tin nhắn từ cậu! 😘❤️";
            document.getElementById('noBtn').style.display = 'none';
            document.getElementById('yesBtn').style.display = 'none';
            
            // Bùng nổ hiệu ứng tim rơi liên tục
            setInterval(createHeart, 40);
        }
    </script>
